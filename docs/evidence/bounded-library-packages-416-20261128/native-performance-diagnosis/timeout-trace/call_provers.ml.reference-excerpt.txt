438: let actualcommand ~config command limit file =
439:   let stime = string_of_int (Float.to_int (Float.ceil limit.limit_time)) in
440:   let stimef = string_of_float limit.limit_time in
441:   let stimems = string_of_int (Float.to_int (limit.limit_time *. 1000.)) in
442:   let smem = string_of_int limit.limit_mem in
443:   let arglist = Cmdline.cmdline_split command in
444:   let use_stdin = ref true in
445:   let on_timelimit = ref false in
446:   let cmd_regexp = Re.Str.regexp "%\\([.]?.\\)" in
447:   let replace s = match Re.Str.matched_group 1 s with
448:     | "%" -> "%"
449:     | "f" -> use_stdin := false; file
450:     | "t" -> on_timelimit := true; stime
451:     | ".t" -> on_timelimit := true; stimef
452:     | "T" -> on_timelimit := true; stimems
453:     | "m" -> smem
454:     | "l" -> Whyconf.libdir config
455:     | "d" -> Whyconf.datadir config
456:     | "S" -> string_of_int limit.limit_steps
457:     | _ -> failwith "unknown specifier, use %%, %f, %t, %.t, %T, %m, %l, %d or %S"
458:   in
459:   let args =
460:     List.map (Re.Str.global_substitute cmd_regexp replace) arglist
461:   in
462:   args, !use_stdin, !on_timelimit
463:
464: let actualcommand ~cleanup ~inplace ~config command limit file =
465:   try
466:     let (cmd, _,_) as x =
467:       actualcommand ~config command limit file

481: let adapt_limits limit on_timelimit =
482:   if limit.limit_time = empty_limits.limit_time then limit
483:   else
484:     { limit with limit_time =
485:       (* for steps limit use 2 * t + 1 time *)
486:       if limit.limit_steps <> empty_limits.limit_steps

489:       else if on_timelimit then 4. *. limit.limit_time +. 1.

491:       else limit.limit_time }
